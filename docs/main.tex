\begin{frame}{Introduction}
\vspace*{-0.75cm} 
\begin{columns}[T,onlytextwidth]
\column{0.58\textwidth}

\begin{procard}
\small
\justifying
The \textbf{Online Canteen Management System} is a web-based platform for campus food ordering. It allows users to browse menu items, place orders, make payments, and track order status digitally while helping the administrator manage menu, stock, and orders efficiently.
\end{procard}

\vspace{0.12cm}

\begin{procard}
\small
\textbf{Target Users}

\vspace{0.05cm}
\begin{itemize}
    \setlength{\itemsep}{2pt}
    \setlength{\parskip}{0pt}
    \item Students
    \item Faculty and staff
    \item Canteen administrator
    \item Kitchen staff
\end{itemize}
\end{procard}

\column{0.38\textwidth}

\begin{procard}
\centering
{\Large\color{Accent}\faUtensils}\\[0.1cm]

{\scriptsize\color{Muted} Faster and easier food requests}

\begin{procard}
\centering
{\Large\color{Success}\faClock}\\[0.1cm]
